{# Scientific sources in APA style (app/apa.py; Ralf, 2026-10-05): in the running text only the short reference cite(...) such as "(Conrad et al., 2017)", linked to its entry; at the end of the page or block one closed fold with the reference list, sorted by author, formatted after APA 7, DOI as link. A click on a short reference opens the fold (app.js). Citations are full strings next to the logic (quality.SOURCES, tempo.CONRAD, ...). #} {# the short reference: "(Conrad et al., 2017; Greszki et al., 2014)"; narrative: "Conrad et al. (2017)" #} {% macro cite(sources, narrative=false, link=true) %}{{ apa.cite(sources, narrative, link) }}{% endmacro %} {# one entry of a list; anchor=false inside a fold that is not the page's list (no second id on the page) #} {% macro entry(source, anchor=true) %}{% set r = apa.ref(source) %}{{ r.html }}{% endmacro %} {% macro _list(entries, uses, use_label=none) %}
    {% for e in entries %}
  • {{ e.html }}{% if uses and e.uses %} {{ use_label or de_en("zu", "on") }}: {{ e.uses|join("; ") }}{% endif %}
    • {% endfor %}
{% endmacro %} {# One fold for everything on a page: groups = [(what the sources belong to, [full citations])]. Each work appears once, sorted by author; what it is cited for stands quietly after it. #} {% macro grouped(groups, label=none, use_label=none) %} {% set entries = apa.references(groups) %}
{{ label or de_en("Literatur", "References") }} ({{ entries|length }}) {{ _list(entries, true, use_label) }}
{% endmacro %} {# a fold with a plain list of citations #} {% macro sources(sources, label=None) %} {% set entries = apa.references(sources) %}
